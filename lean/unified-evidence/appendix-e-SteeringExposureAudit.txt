  0 ≤ duration →
    (path : ℝ → State) →

        Measurable control →
          (∀ s ∈ Icc 0 duration, ‖control s‖ ≤ 1) →

              IntervalIntegrable (fun s => vf (path s) (control s)) volume 0 duration →
