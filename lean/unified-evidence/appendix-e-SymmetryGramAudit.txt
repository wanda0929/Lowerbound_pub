  (∀ x ∈ S,
      ∀ (t : FaceRestriction.Parameters),
        ∑ k,
            (starRingEnd ℂ) (∑ i, A k i * (MvPolynomial.eval (rotatePoint t x)) (QuotientNormalForm.term (a i) 1)) *

@real_invariant_gram : ∀ {ι : Type u_1} {Ω : Type u_2} [inst : Fintype ι] [DecidableEq ι] (q : ι → Charge)
  (odd : ι → Bool) (Q : Matrix ι ι ℂ),
  Q.PosSemidef →
    ∀ (v : Ω → ι → ℂ) (p : Ω → ℝ),
      (∀ (x : Ω) (t : FaceRestriction.Parameters), (value Q fun i => phase (q i) t * v x i) = ↑(p x)) →
        (∀ (x : Ω) (t : FaceRestriction.Parameters),
            (value Q fun i => phase (q i) t * ((if odd i = true then -1 else 1) * (starRingEnd ℂ) (v x i))) = ↑(p x)) →
          ∃ B,
            B.PosSemidef ∧

                ∀ (x : Ω), (value (fun i j => ↑(B i j)) fun i => rephaser (odd i) * v x i) = ↑(p x)
