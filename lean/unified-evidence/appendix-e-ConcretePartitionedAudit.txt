def CZExposure.ConcretePartitioned.Feasible : ExposurePartition → ℝ → ℕ → (ℕ → CellPoly) → Prop :=
fun P U d W =>
  ∀ j < P.J,
    totalDegree (W j) ≤ 2 * d - 1 ∧
      (∀ (ξ : ℝ), IsGInvariant fun x => evalCell ξ x (W j)) ∧
        OccupationCone d (QuotientCertificate.embed (cpoly + cellOp (1 / (U * P.width j)) (W j))) ∧
          (j + 1 < P.J →
              GramCone (interfaceBasis d) (QuotientCertificate.embed (ofState (atXi 0 (W (j + 1)) - atXi 1 (W j))))) ∧
            TerminalCone d (-terminalRestriction (W j))
