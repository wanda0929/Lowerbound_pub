PositiveExposureSegment.mk : (a b : ℝ) →
  a < b →
    (cutoff : ℝ) →
      0 < cutoff →
        (y : ℝ → ExposureEuclidean) →
          (lipConst : NNReal) →
            LipschitzWith lipConst y →
              (u : ℝ → ℂ) →
                Measurable u →
                  (∀ (t : ℝ), ‖u t‖ ≤ 1) →
                    (∀ t ∈ Set.Icc a b, y t ∈ euclideanExposureDomain) →
                      (∀ t ∈ Set.Icc a b, cutoff ≤ euclideanExposureCost (y t)) →
                        (∀ᵐ (t : ℝ) ∂MeasureTheory.volume.restrict (Set.Icc a b),
                            HasDerivAt y (euclideanBfield (y t) (u t)) t) →
                          PositiveExposureSegment

  s ∈ Set.Icc 0 S.duration →
    ∀ (i : Fin 8),
      S.physicalPath s i = S.physicalPath 0 i + ∫ (r : ℝ) in 0..s, vf (S.physicalPath r) (S.physicalControl r) i

  s ∈ Set.Icc 0 S.duration → ∫ (r : ℝ) in 0..s, cost (S.physicalPath r) = S.clockInverseReal s - S.a
PositiveExposureSegment.physicalExposure_eq : ∀ (S : PositiveExposureSegment), S.physicalExposure = S.b - S.a
PositiveExposureSegment.exists_physical_realization : ∀ (S : PositiveExposureSegment),
  ∃ D,
    0 < D ∧
      D ≤ (S.b - S.a) / S.cutoff ∧
        ∃ x u,
          (∃ K, LipschitzWith K x) ∧
            Measurable u ∧
              (∀ (s : ℝ), ‖u s‖ ≤ 1) ∧
                (∀ s ∈ Set.Icc 0 D, x s ∈ Xset) ∧
                  x 0 = S.liftedState S.a ∧
                    x D = S.liftedState S.b ∧
                      (∀ (i : Fin 8), IntervalIntegrable (fun s => vf (x s) (u s) i) MeasureTheory.volume 0 D) ∧
                        (∀ s ∈ Set.Icc 0 D, ∀ (i : Fin 8), x s i = x 0 i + ∫ (r : ℝ) in 0..s, vf (x r) (u r) i) ∧
                          ∫ (s : ℝ) in 0..D, cost (x s) = S.b - S.a
