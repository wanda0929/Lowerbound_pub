def CZExposure.EuclideanCurrentSuperposition : Prop :=
∀ (T : EuclideanNormalCurrentData) (source sink : ExposureEuclidean),
  source ≠ sink →
    T.boundaryPlus = MeasureTheory.Measure.dirac sink →
      T.boundaryMinus = MeasureTheory.Measure.dirac source →
        (∃ K, IsCompact K ∧ T.vectorMeasure.variation Kᶜ = 0) → ∃ A mA D, Nonempty D.CurrentIdentity

ConditionalAcceptance.relaxed_equality : EuclideanCurrentSuperposition → pRelaxed = Cstar
ConditionalAcceptance.smooth_bounds : EuclideanCurrentSuperposition →
  ∀ (a : ℝ),
    ↑a < Cstar →
      ∃ φ,
        ContDiff ℝ (↑⊤) φ ∧
          HasCompactSupport φ ∧
            (∀ x ∈ Xset, ∀ (u : ℂ), ‖u‖ ≤ 1 → |(fderiv ℝ (fun x' => φ (Jmap x') - φ yPi) x) (vf x u)| ≤ cost x) ∧
              (∀ x ∈ XCZ, φ (Jmap x) - φ yPi = 0) ∧ a < φ (Jmap x0) - φ yPi
ConditionalAcceptance.certificates : EuclideanCurrentSuperposition →
  ∀ {ε : ℝ}, 0 < ε → ∃ d W, 1 ≤ d ∧ StationaryFeasible d W ∧ Cstar.toReal - ε < (MvPolynomial.eval x0) W
ConditionalAcceptance.stationary : EuclideanCurrentSuperposition →
  (∀ (d : ℕ), 1 ≤ d → 0 ≤ Lunsplit d ∧ Lunsplit d ≤ Lunsplit (d + 1) ∧ Lunsplit (d + 1) ≤ Cstar) ∧
    Filter.Tendsto Lunsplit Filter.atTop (nhds Cstar)
ConditionalAcceptance.partitioned : EuclideanCurrentSuperposition →
  ∀ (P : ExposurePartition) (U : ℝ),
    (∃ γ, γ.exposure < U) →
      (∀ (d : ℕ),
          1 ≤ d →
            0 ≤ Lunsplit d ∧
              Lunsplit d ≤ Lsplit P U d ∧ Lsplit P U d ≤ Lsplit P U (d + 1) ∧ Lsplit P U (d + 1) ≤ Cstar) ∧
        Filter.Tendsto (Lsplit P U) Filter.atTop (nhds Cstar)
ConditionalAcceptance.concrete_stationary : EuclideanCurrentSuperposition →
  ∀ {ε : ℝ},
    0 < ε →
      ∃ D,
        1 ≤ D ∧
          ∀ d ≥ D,
            0 ≤ Cstar.toReal - (ConcreteStationary.optimum d).toReal ∧
              Cstar.toReal - (ConcreteStationary.optimum d).toReal < ε
ConditionalAcceptance.concrete_partitioned : EuclideanCurrentSuperposition →
  ∀ (P : ExposurePartition) (U : ℝ),
    (∃ γ, γ.exposure < U) → Filter.Tendsto (ConcretePartitioned.optimum P U) Filter.atTop (nhds Cstar)
