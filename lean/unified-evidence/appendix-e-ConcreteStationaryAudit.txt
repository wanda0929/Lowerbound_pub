def CZExposure.ConcreteStationary.Feasible : ℕ → StatePoly → Prop :=
fun d W =>
  totalDegree W ≤ 2 * d - 1 ∧
    (IsGInvariant fun x => (eval x) W) ∧
      ∃ Q T,
        (∀ (q : ↥(ConjugateGram.retained d)), (Q q).PosSemidef) ∧
          T.PosSemidef ∧
            (QuotientCertificate.embed (cpoly + lieDeriv W) -
                  ∑ q,
                    QuotientCertificate.gram ((Q q).map Complex.ofReal) fun i =>
                      C (SymmetryGram.rephaser (decide (Odd (↑i 1 + ↑i 2)))) * QuotientNormalForm.term (⇑↑i) 1) ∈
                QuotientNormalForm.relations ∧
              -terminalRestriction ((rename some) W) = ∑ i, ∑ j, monomialColumn d i * C (T i j) * monomialColumn d j
