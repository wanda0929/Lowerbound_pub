@FiniteExcursionSelection.mk : {T α β : ℝ} →
  {y : ℝ → ExposureEuclidean} →
    {u : ℝ → ℂ} →
      (count : ℕ) →
        (segment : Fin count → PositiveExposureSegment) →
          (∀ (i : Fin count), 0 ≤ (segment i).a ∧ (segment i).b ≤ T) →
            (∀ (i : Fin count), (segment i).y = y) →
              (∀ (i : Fin count), (segment i).u = u) →
                (∀ (i : Fin count), (segment i).cutoff = α) →
                  (∀ (i : Fin count),
                      euclideanExposureCost (y (segment i).a) = α ∧ euclideanExposureCost (y (segment i).b) = α) →
                    (∀ (i : Fin count), ∀ t ∈ Set.Ioo (segment i).a (segment i).b, α < euclideanExposureCost (y t)) →
                      (∀ (i : Fin count), ∃ p ∈ Set.Ioo (segment i).a (segment i).b, β ≤ euclideanExposureCost (y p)) →
                        (Pairwise fun i j =>
                            Disjoint (Set.Ioo (segment i).a (segment i).b) (Set.Ioo (segment j).a (segment j).b)) →
                          (∀ t ∈ Set.Icc 0 T,
                              β ≤ euclideanExposureCost (y t) → ∃ i, t ∈ Set.Ioo (segment i).a (segment i).b) →
                            FiniteExcursionSelection T α β y u
@FiniteExcursionSelection.ofTrajectory : {T α β : ℝ} →
  {y : ℝ → ExposureEuclidean} →
    {u : ℝ → ℂ} →
      {K : NNReal} →
        0 ≤ T →
          0 < α →
            α < β →
              LipschitzWith K y →
                Measurable u →
                  (∀ (t : ℝ), ‖u t‖ ≤ 1) →
                    (∀ t ∈ Set.Icc 0 T, y t ∈ euclideanExposureDomain) →
                      (∀ᵐ (t : ℝ) ∂MeasureTheory.volume.restrict (Set.Icc 0 T),
                          HasDerivAt y (euclideanBfield (y t) (u t)) t) →
                        euclideanExposureCost (y 0) ≤ α →
                          euclideanExposureCost (y T) ≤ α → FiniteExcursionSelection T α β y u

@FiniteExcursionSelection.count_bound : ∀ {T α β : ℝ} {y : ℝ → ExposureEuclidean} {u : ℝ → ℂ}
  (S : FiniteExcursionSelection T α β y u),
  0 ≤ T →
    0 ≤ α →
      α < β →
        LipschitzOnWith (3 / 2) (fun t => costThreeHalves (euclideanExposureCost (y t))) (Set.Icc 0 T) →
          ↑S.count * (4 / 3 * (costThreeHalves β - costThreeHalves α)) ≤ T
