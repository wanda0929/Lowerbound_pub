@finite_low_cost_phase_accounting : ∀ {A : Type u_1} (I : Finset A) (l r : A → ℝ) {T β : ℝ} {y : ℝ → ExposureEuclidean}
  {u : ℝ → ℂ} {K : NNReal},
  0 ≤ T →
    LipschitzWith K y →
      Measurable u →
        (∀ (t : ℝ), ‖u t‖ ≤ 1) →
          (∀ t ∈ Icc 0 T, y t ∈ euclideanExposureDomain) →
            (∀ᵐ (t : ℝ) ∂volume.restrict (Icc 0 T), HasDerivAt y (euclideanBfield (y t) (u t)) t) →
              β ≤ 1 / 12 →
                (∀ i ∈ I, 0 ≤ l i ∧ l i ≤ r i ∧ r i ≤ T) →
                  ((↑I).PairwiseDisjoint fun i => Ioo (l i) (r i)) →
                    (∀ i ∈ I, ∀ t ∈ Icc (l i) (r i), euclideanExposureCost (y t) ≤ β) →
                      ∑ i ∈ I, |exposurePhaseIncrement y u (l i) (r i)| ≤ 8 * T * √β ∧
                        |∑ i ∈ I, exposurePhaseIncrement y u (l i) (r i)| ≤ 8 * T * √β ∧
                          ∑ i ∈ I, exposurePhaseIncrement y u (l i) (r i) =
                              ∫ (t : ℝ) in ⋃ i ∈ I, Ioo (l i) (r i), euclideanAngularRate (y t) (u t) ∧
                            ∏ i ∈ I,
                                  complexUnitPhase (exposureProgressPath y (r i)) /
                                    complexUnitPhase (exposureProgressPath y (l i)) =
                                phaseFactor (∑ i ∈ I, exposurePhaseIncrement y u (l i) (r i)) ∧
                              ‖∏ i ∈ I,
                                      complexUnitPhase (exposureProgressPath y (r i)) /
                                        complexUnitPhase (exposureProgressPath y (l i)) -
                                    1‖ ≤
                                8 * T * √β

                    (∀ i ∈ I, Icc (l i) (r i) ⊆ S.discardedTimes) →
