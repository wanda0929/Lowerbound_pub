EuclideanNormalCurrentData.mk : (vectorMeasure : MeasureTheory.VectorMeasure ExposureEuclidean ExposureEuclidean) →
  MeasureTheory.IsFiniteMeasure vectorMeasure.variation →
    (direction : ExposureEuclidean → ExposureEuclidean) →
      Measurable direction →
        MeasureTheory.Integrable direction vectorMeasure.variation →
          (∀ᵐ (z : ExposureEuclidean) ∂vectorMeasure.variation, ‖direction z‖ = 1) →
            vectorMeasure = vectorMeasure.variation.withDensityᵥ direction →
              (boundaryPlus : MeasureTheory.Measure ExposureEuclidean) →
                MeasureTheory.IsFiniteMeasure boundaryPlus →
                  (boundaryMinus : MeasureTheory.Measure ExposureEuclidean) →
                    MeasureTheory.IsFiniteMeasure boundaryMinus →
                      (∀ (ψ : ExposureEuclidean → ℝ),
                          ContDiff ℝ (↑⊤) ψ →
                            HasCompactSupport ψ →
                              MeasureTheory.Integrable (fun z => (fderiv ℝ ψ z) (direction z)) vectorMeasure.variation ∧
                                ∫ (z : ExposureEuclidean), (fderiv ℝ ψ z) (direction z) ∂vectorMeasure.variation =
                                  ∫ (z : ExposureEuclidean), ψ z ∂boundaryPlus -
                                    ∫ (z : ExposureEuclidean), ψ z ∂boundaryMinus) →
                        EuclideanNormalCurrentData
