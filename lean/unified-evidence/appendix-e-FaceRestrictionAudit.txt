@stationary_factor_mem_reduced_span : ∀ {χ : Type u_1} {ι : Type u_2} [Fintype χ] [DecidableEq χ] [Fintype ι]
  (keys : χ → Index),
  Function.Injective keys →
    ∀ (z : ι → ComplexPoly) (m : χ → ℕ → ComplexPoly) (n : χ → ℕ),
      (∀ (i : ι), restriction (z i) = 0) →
