  0 < α →
    (∀ x ∈ Xset,
        ∀ (u : ℂ), ‖u‖ ≤ 1 → ∀ (ξ : ℝ), α * cost x ≤ (eval (pt ξ x u)) (cpoly + lieDeriv (strictPolynomial P η α))) →
      ∃ A,

          amplitudes ⬝ᵥ A *ᵥ amplitudes = cpoly + lieDeriv (strictPolynomial P η α) ∧
