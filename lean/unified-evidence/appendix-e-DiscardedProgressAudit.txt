    LipschitzWith K y →
      Measurable u →
        (∀ (t : ℝ), ‖u t‖ ≤ 1) →
          (∀ t ∈ Set.Icc 0 T, y t ∈ euclideanExposureDomain) →
            (∀ᵐ (t : ℝ) ∂MeasureTheory.volume.restrict (Set.Icc 0 T), HasDerivAt y (euclideanBfield (y t) (u t)) t) →
              (∀ i ∈ I, 0 ≤ l i ∧ l i ≤ r i ∧ r i ≤ T) →
