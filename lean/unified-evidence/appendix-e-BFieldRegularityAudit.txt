CZExposure.Bfield_norm_le_sqrt_cost : ∃ K,
  0 < K ∧
    ∀ x ∈ CZExposure.Xset, ∀ (u : ℂ), ‖u‖ ≤ 1 → ‖CZExposure.Bfield (CZExposure.Jmap x) u‖ ≤ K * √(CZExposure.cost x)
